\CPTitleBlock{Roots of Polynomials}{Problem Sheet 4 \enspace$\cdot$\enspace Vieta's Relations, Symmetric Functions \& Root Transformations}

\section*{Section A \quad Procedural \& Direct Foundations \hfill{\normalfont\footnotesize(Questions 1--3)}}
\begin{enumerate}

%── Q1 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} The cubic equation
\[
2x^3 - 5x^2 + 8x - 4 = 0
\]
has roots $\alpha$, $\beta$, and $\gamma$. Without solving the equation, state the exact values of:
\begin{enumerate}[label=(\alph*)]
  \item $\alpha + \beta + \gamma$ \hfill{\textbf{[1 mark]}}
  \item $\alpha\beta + \beta\gamma + \gamma\alpha$ \hfill{\textbf{[1 mark]}}
  \item $\alpha\beta\gamma$ \hfill{\textbf{[1 mark]}}
  \item Hence find the exact value of $\alpha^2 + \beta^2 + \gamma^2$. \hfill{\textbf{[2 marks]}}
\end{enumerate}

%── Q2 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} The quartic equation
\[
x^4 + 3x^3 - 6x^2 - 8x + 2 = 0
\]
has roots $\alpha$, $\beta$, $\gamma$, and $\delta$.
\begin{enumerate}[label=(\alph*)]
  \item Write down the values of $\sum\alpha$, $\sum\alpha\beta$, $\sum\alpha\beta\gamma$, and $\alpha\beta\gamma\delta$. \hfill{\textbf{[2 marks]}}
  \item Find the exact value of $\alpha^2 + \beta^2 + \gamma^2 + \delta^2$. \hfill{\textbf{[2 marks]}}
  \item Find the value of $(\alpha + 1)(\beta + 1)(\gamma + 1)(\delta + 1)$. \hfill{\textbf{[2 marks]}}
\end{enumerate}

%── Q3 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} The roots of the cubic equation $x^3 - 4x^2 + 6x - 2 = 0$ are $\alpha$, $\beta$, and $\gamma$.
\begin{enumerate}[label=(\alph*)]
  \item Find the exact value of $\dfrac{1}{\alpha} + \dfrac{1}{\beta} + \dfrac{1}{\gamma}$. \hfill{\textbf{[2 marks]}}
  \item Show that $\dfrac{1}{\alpha^2} + \dfrac{1}{\beta^2} + \dfrac{1}{\gamma^2} = 5$. \hfill{\textbf{[3 marks]}}
\end{enumerate}

\end{enumerate}

\section*{Section B \quad Reverse Parameters \& Analysis \hfill{\normalfont\footnotesize(Questions 4--6)}}
\begin{enumerate}[resume]

%── Q4 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} The cubic equation $x^3 - 7x^2 + kx - 8 = 0$ has roots $\alpha$, $\beta$, and $\gamma$.
Given that the three roots are in geometric progression such that $\beta^2 = \alpha\gamma$:
\begin{enumerate}[label=(\alph*)]
  \item Find the exact integer value of the middle root $\beta$. \hfill{\textbf{[2 marks]}}
  \item Hence determine the values of the other two roots $\alpha$ and $\gamma$. \hfill{\textbf{[2 marks]}}
  \item Find the exact value of the real constant $k$. \hfill{\textbf{[2 marks]}}
\end{enumerate}

%── Q5 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} The roots of the cubic equation $x^3 - 3x^2 + 5x - 7 = 0$ are $\alpha$, $\beta$, and $\gamma$.
\begin{enumerate}[label=(\alph*)]
  \item Explain why $\alpha^3 - 3\alpha^2 + 5\alpha - 7 = 0$. \hfill{\textbf{[1 mark]}}
  \item By summing this relation over all three roots, prove that
  \[
  \alpha^3 + \beta^3 + \gamma^3 = 3(\alpha^2 + \beta^2 + \gamma^2) - 5(\alpha + \beta + \gamma) + 21
  \]
  \hfill{\textbf{[2 marks]}}
  \item Hence find the exact integer value of $\alpha^3 + \beta^3 + \gamma^3$. \hfill{\textbf{[3 marks]}}
\end{enumerate}

%── Q6 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} The cubic equation $x^3 + px^2 + qx - 48 = 0$ has roots $\alpha$, $\beta$, and $\gamma$, where $p$ and $q$ are real constants.
Given that the roots are in the ratio $1 : 2 : 3$:
\begin{enumerate}[label=(\alph*)]
  \item Determine the values of $\alpha$, $\beta$, and $\gamma$. \hfill{\textbf{[3 marks]}}
  \item Hence find the values of the constants $p$ and $q$. \hfill{\textbf{[3 marks]}}
\end{enumerate}

\end{enumerate}

\section*{Section C \quad Proof, Geometry \& Multi-step \hfill{\normalfont\footnotesize(Questions 7--10)}}
\begin{enumerate}[resume]

%── Q7 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} The cubic equation $x^3 - 2x^2 + 4x - 5 = 0$ has roots $\alpha$, $\beta$, and $\gamma$.
\begin{enumerate}[label=(\alph*)]
  \item Use the substitution $w = x + 2$ to obtain a cubic equation in $w$ whose roots are $\alpha + 2$, $\beta + 2$, and $\gamma + 2$. Give your answer in the form $w^3 + Aw^2 + Bw + C = 0$, where $A, B, C$ are integers. \hfill{\textbf{[4 marks]}}
  \item Verify your value of $C$ by evaluating $(\alpha + 2)(\beta + 2)(\gamma + 2)$ using Vieta's formulas directly. \hfill{\textbf{[2 marks]}}
\end{enumerate}

%── Q8 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} The roots of the cubic equation $2x^3 - 3x^2 + 4x - 1 = 0$ are $\alpha$, $\beta$, and $\gamma$.
\begin{enumerate}[label=(\alph*)]
  \item Using the substitution $w = 2x - 1$, find a cubic equation with integer coefficients whose roots are $2\alpha - 1$, $2\beta - 1$, and $2\gamma - 1$. \hfill{\textbf{[4 marks]}}
  \item State the value of the product $(2\alpha - 1)(2\beta - 1)(2\gamma - 1)$. \hfill{\textbf{[1 mark]}}
\end{enumerate}

%── Q9 [SYN] ─────────────────────────────────────────────────────────────
\item \textbf{[SYN]} The cubic equation $x^3 - 3x + 1 = 0$ has roots $\alpha$, $\beta$, and $\gamma$.
\begin{enumerate}[label=(\alph*)]
  \item By writing $x(x^2 - 3) = -1$ and squaring both sides, or using the substitution $w = x^2$, show that the cubic equation whose roots are $\alpha^2$, $\beta^2$, and $\gamma^2$ is
  \[
  w^3 - 6w^2 + 9w - 1 = 0
  \]
  \hfill{\textbf{[4 marks]}}
  \item Hence deduce the exact values of:
  \begin{enumerate}[label=(\roman*)]
    \item $\alpha^2 + \beta^2 + \gamma^2$ \hfill{\textbf{[1 mark]}}
    \item $\alpha^4 + \beta^4 + \gamma^4$ \hfill{\textbf{[2 marks]}}
  \end{enumerate}
\end{enumerate}

%── Q10 [SYN] ────────────────────────────────────────────────────────────
\item \textbf{[SYN]} The quartic equation $x^4 - 2x^3 + 3x^2 - 4x + 1 = 0$ has roots $\alpha, \beta, \gamma, \delta$.
\begin{enumerate}[label=(\alph*)]
  \item Use the substitution $w = \dfrac{1}{x^2}$ to derive a quartic polynomial in $w$ whose roots are $\dfrac{1}{\alpha^2}, \dfrac{1}{\beta^2}, \dfrac{1}{\gamma^2}, \dfrac{1}{\delta^2}$. \hfill{\textbf{[5 marks]}}
  \item Hence determine the exact value of $\dfrac{1}{\alpha^2} + \dfrac{1}{\beta^2} + \dfrac{1}{\gamma^2} + \dfrac{1}{\delta^2}$. \hfill{\textbf{[2 marks]}}
\end{enumerate}

\end{enumerate}
