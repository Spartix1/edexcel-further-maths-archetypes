\section*{Section A \quad Procedural \& Direct Foundations}
\begin{enumerate}

%── Q1 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} Cubic Vieta relations and quadratic symmetric sum.
\ans{
(a) $\dfrac{5}{2}$ \qquad (b) $4$ \qquad (c) $2$ \qquad (d) $-\dfrac{7}{4}$
}
\method{
\textbf{(a)} B1: $\sum\alpha = -\frac{-5}{2} = \frac{5}{2}$.\\
\textbf{(b)} B1: $\sum\alpha\beta = \frac{8}{2} = 4$.\\
\textbf{(c)} B1: $\alpha\beta\gamma = -\frac{-4}{2} = 2$.\\
\textbf{(d)} M1: Uses identity $\alpha^2 + \beta^2 + \gamma^2 = (\sum\alpha)^2 - 2\sum\alpha\beta$.
M1: Substitutes: $\left(\frac{5}{2}\right)^2 - 2(4) = \frac{25}{4} - 8$.
A1: $-\frac{7}{4}$.
}
\inv{
Why is $\alpha^2 + \beta^2 + \gamma^2$ negative ($-\frac{7}{4}$)? Recall that for any real numbers, the sum of squares must be non-negative. What does a negative sum of squares immediately reveal about the roots of this polynomial? (It proves that at least two roots must be non-real complex numbers!).
}

%── Q2 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} Quartic Vieta relations and factored bracket evaluations.
\ans{
(a) $\sum\alpha = -3, \ \sum\alpha\beta = -6, \ \sum\alpha\beta\gamma = 8, \ \alpha\beta\gamma\delta = 2$ \quad (b) $21$ \quad (c) $2$
}
\method{
\textbf{(a)} B1: $\sum\alpha = -3$ and $\sum\alpha\beta = -6$.
B1: $\sum\alpha\beta\gamma = -(-8) = 8$ and $\alpha\beta\gamma\delta = 2$.\\
\textbf{(b)} M1: Uses $\sum\alpha^2 = (\sum\alpha)^2 - 2\sum\alpha\beta = (-3)^2 - 2(-6) = 9 + 12$.
A1: $21$.\\
\textbf{(c)} M1: Recognizes polynomial factorisation $P(x) = (x-\alpha)(x-\beta)(x-\gamma)(x-\delta) = x^4 + 3x^3 - 6x^2 - 8x + 2$.
M1: Evaluates at $x = -1$:
\[
P(-1) = (-1-\alpha)(-1-\beta)(-1-\gamma)(-1-\delta) = (\alpha+1)(\beta+1)(\gamma+1)(\delta+1)
\]
A1: $P(-1) = (-1)^4 + 3(-1)^3 - 6(-1)^2 - 8(-1) + 2 = 1 - 3 - 6 + 8 + 2 = 2$.
}
\inv{
Notice how fast evaluating $P(-1)$ is compared to expanding $(\alpha+1)(\beta+1)(\gamma+1)(\delta+1) = \alpha\beta\gamma\delta + \sum\alpha\beta\gamma + \sum\alpha\beta + \sum\alpha + 1$. Verify that substituting the Vieta values into the expanded expression yields $2 + 8 + (-6) + (-3) + 1 = 2$.
}

%── Q3 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} Reciprocal symmetric expressions for cubics.
\ans{
(a) $3$ \qquad (b) Proof showing $\sum \dfrac{1}{\alpha^2} = 5$
}
\method{
\textbf{(a)} M1: $\sum\alpha = 4, \ \sum\alpha\beta = 6, \ \alpha\beta\gamma = 2$.
A1: $\sum\frac{1}{\alpha} = \frac{\sum\alpha\beta}{\alpha\beta\gamma} = \frac{6}{2} = 3$.\\
\textbf{(b)} M1: Uses identity $\sum\frac{1}{\alpha^2} = \left(\sum\frac{1}{\alpha}\right)^2 - 2\sum\frac{1}{\alpha\beta}$.
M1: $\sum\frac{1}{\alpha\beta} = \frac{\sum\alpha}{\alpha\beta\gamma} = \frac{4}{2} = 2$.
A1: $\sum\frac{1}{\alpha^2} = 3^2 - 2(2) = 9 - 4 = 5$.
}
\inv{
Could you solve this by creating the reciprocal equation? Let $y = \frac{1}{x} \implies x = \frac{1}{y}$. Substituting into $x^3 - 4x^2 + 6x - 2 = 0$ gives $-2y^3 + 6y^2 - 4y + 1 = 0 \implies 2y^3 - 6y^2 + 4y - 1 = 0$. Now apply Vieta's formulas directly to $y$: $\sum y = \frac{6}{2} = 3$ and $\sum y^2 = 3^2 - 2\left(\frac{4}{2}\right) = 5$!
}

\end{enumerate}

\section*{Section B \quad Reverse Parameters \& Analysis}
\begin{enumerate}[resume]

%── Q4 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} Roots in geometric progression and parameter solving.
\ans{
(a) $\beta = 2$ \qquad (b) $\alpha = 1, \quad \gamma = 4 \quad (\text{or vice versa})$ \qquad (c) $k = 14$
}
\method{
\textbf{(a)} M1: By Vieta, $\alpha\beta\gamma = 8$. Given $\beta^2 = \alpha\gamma$, substitute: $\beta^2 \times \beta = \beta^3 = 8$.
A1: Since $\beta \in \mathbb{R}$, $\beta = 2$.\\
\textbf{(b)} M1: Sum of roots $\alpha + \beta + \gamma = 7 \implies \alpha + \gamma = 7 - 2 = 5$.
Product $\alpha\gamma = \beta^2 = 4$.
A1: Roots of $t^2 - 5t + 4 = 0$ are $(t-1)(t-4) = 0 \implies \alpha = 1, \gamma = 4$.\\
\textbf{(c)} M1: By Vieta, $k = \sum\alpha\beta = \alpha\beta + \beta\gamma + \gamma\alpha$.
A1: $k = (1)(2) + (2)(4) + (1)(4) = 2 + 8 + 4 = 14$.
}
\inv{
What if the three roots were in arithmetic progression ($\alpha + \gamma = 2\beta$)? Prove that for any cubic $x^3 + px^2 + qx + r = 0$, the roots are in arithmetic progression if and only if $2p^3 - 9pq + 27r = 0$.
}

%── Q5 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} High power symmetric sums via polynomial recurrence.
\ans{
(a) Root substitution property \qquad (b) Summation identity derivation \qquad (c) $\alpha^3 + \beta^3 + \gamma^3 = 3$
}
\method{
\textbf{(a)} B1: Since $\alpha$ is a root of the equation, substituting $x = \alpha$ gives $\alpha^3 - 3\alpha^2 + 5\alpha - 7 = 0$.\\
\textbf{(b)} M1: Writes the same identity for $\beta$ and $\gamma$ and adds all three equations:
\[
(\alpha^3 + \beta^3 + \gamma^3) - 3(\alpha^2 + \beta^2 + \gamma^2) + 5(\alpha + \beta + \gamma) - 3(7) = 0
\]
A1: Rearranges to $\sum\alpha^3 = 3\sum\alpha^2 - 5\sum\alpha + 21$.\\
\textbf{(c)} M1: $\sum\alpha = 3$ and $\sum\alpha\beta = 5$.
$\sum\alpha^2 = (\sum\alpha)^2 - 2\sum\alpha\beta = 3^2 - 2(5) = 9 - 10 = -1$.
M1: Substitutes: $\sum\alpha^3 = 3(-1) - 5(3) + 21 = -3 - 15 + 21$.
A1: $3$.
}
\inv{
Newton's sums state that for $P(x) = x^n + a_1 x^{n-1} + \dots + a_n = 0$, the power sums $S_k = \sum \alpha^k$ satisfy $S_k + a_1 S_{k-1} + a_2 S_{k-2} + \dots = 0$. Use this relation to find $\alpha^4 + \beta^4 + \gamma^4$ for this cubic without expanding brackets!
}

%── Q6 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} Roots in fixed ratio and coefficient determination.
\ans{
(a) $\alpha = 2, \ \beta = 4, \ \gamma = 6$ \qquad (b) $p = -12, \quad q = 44$
}
\method{
\textbf{(a)} M1: Let the roots be $u, 2u, 3u$.
Product of roots: $\alpha\beta\gamma = u(2u)(3u) = 6u^3$.
M1: By Vieta, $\alpha\beta\gamma = -(-48) = 48 \implies 6u^3 = 48 \implies u^3 = 8 \implies u = 2$.
A1: Roots are $\alpha = 2, \beta = 4, \gamma = 6$.\\
\textbf{(b)} M1: Sum of roots: $-p = \alpha + \beta + \gamma = 2 + 4 + 6 = 12 \implies p = -12$.
M1: Pairwise sum: $q = \sum\alpha\beta = (2)(4) + (4)(6) + (2)(6) = 8 + 24 + 12$.
A1: $p = -12$ and $q = 44$.
}
\inv{
Verify by expanding the polynomial factors directly: $(x-2)(x-4)(x-6) = (x^2 - 6x + 8)(x - 6) = x^3 - 6x^2 - 6x^2 + 36x + 8x - 48 = x^3 - 12x^2 + 44x - 48$. This confirms the Vieta deductions with 100\% certainty!
}

\end{enumerate}

\section*{Section C \quad Proof, Geometry \& Multi-step}
\begin{enumerate}[resume]

%── Q7 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} Linear root transformation via substitution.
\ans{
(a) $w^3 - 8w^2 + 24w - 29 = 0$ \qquad (b) Verification via $(\alpha+2)(\beta+2)(\gamma+2) = 29$
}
\method{
\textbf{(a)} M1: Let $w = x + 2 \implies x = w - 2$.
M1: Substitutes into original cubic:
\[
(w - 2)^3 - 2(w - 2)^2 + 4(w - 2) - 5 = 0
\]
M1: Expands systematically:
$(w^3 - 6w^2 + 12w - 8) - 2(w^2 - 4w + 4) + 4w - 8 - 5 = 0$.
A1: $w^3 - 8w^2 + 24w - 29 = 0$ ($A = -8, B = 24, C = -29$).\\
\textbf{(b)} M1: Constant term of $w$ equation is $-29 \implies \text{Product of roots} = -(-29) = 29$.
A1: Using $P(x) = (x-\alpha)(x-\beta)(x-\gamma)$, notice $(\alpha+2)(\beta+2)(\gamma+2) = -P(-2) = -((-2)^3 - 2(-2)^2 + 4(-2) - 5) = -(-8 - 8 - 8 - 5) = -(-29) = 29$, confirming consistency.
}
\inv{
Why does substituting $x = w - h$ shift the graph of $P(x)$ to the right by $h$? In particular, if you choose $h = \frac{2}{3}$, what happens to the $w^2$ term in the transformed equation? (This is the Tschirnhaus transformation that eliminates the quadratic term in a cubic!).
}

%── Q8 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} Multiplicative root transformations.
\ans{
(a) $w^3 + 5w + 2 = 0$ \qquad (b) Product of roots $= -2$
}
\method{
\textbf{(a)} M1: Sets $w = 2x - 1 \implies x = \frac{w+1}{2}$.
M1: Substitutes into $2x^3 - 3x^2 + 4x - 1 = 0$:
\[
2\left(\frac{w+1}{2}\right)^3 - 3\left(\frac{w+1}{2}\right)^2 + 4\left(\frac{w+1}{2}\right) - 1 = 0
\]
M1: Multiplies by 4 to clear denominators:
\[
(w+1)^3 - 3(w+1)^2 + 8(w+1) - 4 = 0
\]
A1: Expands: $(w^3 + 3w^2 + 3w + 1) - 3(w^2 + 2w + 1) + 8w + 8 - 4 = w^3 + 5w + 2 = 0$.\\
\textbf{(b)} B1: Constant term is $2$, so product of roots $= -2$.
}
\inv{
Notice that the $w^2$ coefficient in $w^3 + 5w + 2 = 0$ is zero! This means the sum of the transformed roots is $\sum (2\alpha - 1) = 0$. Check: $2\sum\alpha - 3 = 2\left(\frac{3}{2}\right) - 3 = 0$!
}

%── Q9 [SYN] ─────────────────────────────────────────────────────────────
\item \textbf{[SYN]} Squared root transformation and higher power sums.
\ans{
(a) Proof of $w^3 - 6w^2 + 9w - 1 = 0$ \qquad (b)(i) $6$ \qquad (b)(ii) $18$
}
\method{
\textbf{(a)} M1: Rearranges $x^3 - 3x + 1 = 0$ to $x(x^2 - 3) = -1$.
M1: Squares both sides: $x^2(x^2 - 3)^2 = 1$.
M1: Substitutes $w = x^2$: $w(w - 3)^2 = 1$.
A1: Expands $w(w^2 - 6w + 9) = 1 \implies w^3 - 6w^2 + 9w - 1 = 0$.\\
\textbf{(b)(i)} B1: $\alpha^2 + \beta^2 + \gamma^2 = \sum w = -(-6) = 6$.\\
\textbf{(b)(ii)} M1: $\alpha^4 + \beta^4 + \gamma^4 = \sum w^2 = (\sum w)^2 - 2\sum w_i w_j$.
A1: $6^2 - 2(9) = 36 - 18 = 18$.
}
\inv{
Since $x^3 - 3x + 1 = 0$ has real roots, notice that $x = 2\cos\theta$ transforms it into $8\cos^3\theta - 6\cos\theta + 1 = 0 \implies 2\cos(3\theta) = -1 \implies \cos(3\theta) = -\frac{1}{2}$. The roots are $2\cos\frac{2\pi}{9}, 2\cos\frac{4\pi}{9}, 2\cos\frac{8\pi}{9}$. Thus part (b) proves $\cos^2\frac{2\pi}{9} + \cos^2\frac{4\pi}{9} + \cos^2\frac{8\pi}{9} = \frac{6}{4} = \frac{3}{2}$!
}

%── Q10 [SYN] ────────────────────────────────────────────────────────────
\item \textbf{[SYN]} Quartic reciprocal squared transformation.
\ans{
(a) $w^4 - 10w^3 + 27w^2 - 12w + 1 = 0$ \qquad (b) $10$
}
\method{
\textbf{(a)} M1: First forms reciprocal equation with $y = \frac{1}{x}$:
$1 - 4y + 3y^2 - 2y^3 + y^4 = 0 \implies y^4 - 2y^3 + 3y^2 - 4y + 1 = 0$.
M1: Rearranges into even and odd powers: $y^4 + 3y^2 + 1 = 2y^3 + 4y = 2y(y^2 + 2)$.
M1: Squares both sides: $(y^4 + 3y^2 + 1)^2 = 4y^2(y^2 + 2)^2$.
M1: Let $w = y^2 = \frac{1}{x^2}$:
$(w^2 + 3w + 1)^2 = 4w(w + 2)^2$.
A1: $(w^4 + 6w^3 + 11w^2 + 6w + 1) = 4w(w^2 + 4w + 4) = 4w^3 + 16w^2 + 16w \implies w^4 - 10w^3 + 27w^2 - 12w + 1 = 0$.\\
\textbf{(b)} M1: Sum of roots of the $w$ polynomial is $\sum w = -(-10) = 10$.
A1: $\sum\frac{1}{\alpha^2} = 10$.
}
\inv{
Verify part (b) directly via Vieta's formulas on the original polynomial:
$\sum\frac{1}{\alpha} = \frac{\sum\alpha\beta\gamma}{\alpha\beta\gamma\delta} = \frac{-(-4)}{1} = 4$.
$\sum\frac{1}{\alpha\beta} = \frac{\sum\alpha\beta}{\alpha\beta\gamma\delta} = \frac{3}{1} = 3$.
Then $\sum\frac{1}{\alpha^2} = 4^2 - 2(3) = 16 - 6 = 10$! Both methods converge to $10$.
}

\end{enumerate}
