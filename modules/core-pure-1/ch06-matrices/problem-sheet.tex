\section*{Section A \quad Procedural \& Direct Foundations \hfill{\normalfont\footnotesize(Questions 1--3)}}
\begin{enumerate}
%── Q1 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} The matrix $\mathbf{A}$ is given by
\[
\mathbf{A} = \begin{pmatrix} 2 & 1 & -1 \\ 1 & 3 & 2 \\ 3 & 2 & 1 \end{pmatrix}
\]
\begin{enumerate}
  \item Evaluate $\det(\mathbf{A})$. \hfill{\small\textbf{[2]}}
  \item Given that $\mathbf{B}$ is a non-singular $3 \times 3$ matrix such that $\det(\mathbf{A}\mathbf{B}) = 120$, find:
  \begin{enumerate}[label=(\roman*)]
    \item $\det(\mathbf{B}^{-1})$,
    \item $\det(2\mathbf{B})$. \hfill{\small\textbf{[3]}}
  \end{enumerate}
\end{enumerate}

%── Q2 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} The matrix $\mathbf{M}$ is defined by
\[
\mathbf{M} = \begin{pmatrix} 1 & 2 & 1 \\ 2 & 1 & 1 \\ 1 & 1 & 2 \end{pmatrix}
\]
\begin{enumerate}
  \item Show that $\det(\mathbf{M}) = -4$. \hfill{\small\textbf{[1]}}
  \item Form the matrix of minors and the matrix of cofactors of $\mathbf{M}$. \hfill{\small\textbf{[3]}}
  \item Hence determine the adjugate matrix $\operatorname{adj}(\mathbf{M})$ and write down $\mathbf{M}^{-1}$. \hfill{\small\textbf{[2]}}
\end{enumerate}

%── Q3 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} A system of simultaneous equations is given by
\[
\begin{aligned}
x + 2y + z &= 8 \\
2x + y + z &= 7 \\
x + y + 2z &= 9
\end{aligned}
\]
\begin{enumerate}
  \item Express this system in the matrix form $\mathbf{M}\mathbf{x} = \mathbf{b}$. \hfill{\small\textbf{[1]}}
  \item Using the matrix inverse $\mathbf{M}^{-1}$ obtained in Question~2, find the unique solution $(x, y, z)$. \hfill{\small\textbf{[4]}}
\end{enumerate}

\end{enumerate}

\section*{Section B \quad Reverse Parameters \& Analysis \hfill{\normalfont\small(Questions 4--6)}}
\begin{enumerate}[resume]
%── Q4 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} The matrix $\mathbf{P}$ is parameterized by $k \in \mathbb{R}$:
\[
\mathbf{P} = \begin{pmatrix} 1 & 2 & k \\ 2 & k & 4 \\ 1 & 1 & 1 \end{pmatrix}
\]
\begin{enumerate}
  \item Show that $\det(\mathbf{P}) = k(3 - k)$. \hfill{\small\textbf{[3]}}
  \item State the values of $k$ for which $\mathbf{P}$ is singular. \hfill{\small\textbf{[1]}}
  \item Hence state the set of values of $k$ for which $\mathbf{P}^{-1}$ exists. \hfill{\small\textbf{[1]}}
\end{enumerate}

%── Q5 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]}
\begin{enumerate}
  \item Prove from first principles that if $\mathbf{A}$ and $\mathbf{B}$ are non-singular $n \times n$ matrices, then
  \[
  (\mathbf{A}\mathbf{B})^{-1} = \mathbf{B}^{-1}\mathbf{A}^{-1}
  \]
  \hfill{\small\textbf{[2]}}
  \item The $2 \times 2$ matrices $\mathbf{M}$ and $\mathbf{N}$ satisfy $\mathbf{N}^{-1}\mathbf{M}\mathbf{N} = \begin{pmatrix} 3 & 0 \\ 0 & -1 \end{pmatrix}$.
  \begin{enumerate}[label=(\roman*)]
    \item Find $\det(\mathbf{M})$.
    \item Evaluate $\det(\mathbf{M}^5)$.
    \item Find the matrix $\mathbf{N}^{-1}\mathbf{M}^4\mathbf{N}$. \hfill{\small\textbf{[4]}}
  \end{enumerate}
\end{enumerate}

%── Q6 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} Consider the system of linear equations in $x, y, z$:
\[
\begin{aligned}
x + y + 2z &= 4 \\
2x - y + z &= 1 \\
3x + ky + 3z &= d
\end{aligned}
\]
where $k$ and $d$ are real constants.
\begin{enumerate}
  \item Show that the determinant of the coefficient matrix is $3k$. \hfill{\small\textbf{[2]}}
  \item Given that the system does not have a unique solution:
  \begin{enumerate}[label=(\roman*)]
    \item state the value of $k$,
    \item find the specific value of $d$ for which the system has infinitely many solutions,
    \item describe the geometric arrangement of the three planes when $d$ takes this value, and when $d \neq 5$. \hfill{\small\textbf{[5]}}
  \end{enumerate}
\end{enumerate}

\end{enumerate}

\section*{Section C \quad Proof, Geometry \& Multi-step \hfill{\normalfont\small(Questions 7--10)}}
\begin{enumerate}[resume]
%── Q7 [GEO] ─────────────────────────────────────────────────────────────
\item \textbf{[GEO]} Three planes $\Pi_1$, $\Pi_2$, and $\Pi_3$ have Cartesian equations
\[
\begin{aligned}
\Pi_1 &: x - 2y + z = 3 \\
\Pi_2 &: 2x + y - z = 4 \\
\Pi_3 &: 3x - y = 8
\end{aligned}
\]
\begin{enumerate}
  \item Show that the three planes do not have any common point of intersection. \hfill{\small\textbf{[3]}}
  \item Verify that no two of the planes are parallel. \hfill{\small\textbf{[2]}}
  \item Deduce that the three planes form the faces of a triangular prism, and find a direction vector parallel to the three lines of pairwise intersection. \hfill{\small\textbf{[3]}}
\end{enumerate}

%── Q8 [GEO] ─────────────────────────────────────────────────────────────
\item \textbf{[GEO]} The planes $\Pi_1$, $\Pi_2$, and $\Pi_3$ are defined by
\[
\begin{aligned}
\Pi_1 &: x + 2y - z = 2 \\
\Pi_2 &: 2x - y + 3z = 9 \\
\Pi_3 &: 4x + 3y + z = 13
\end{aligned}
\]
\begin{enumerate}
  \item Show that $\det(\mathbf{M}) = 0$, where $\mathbf{M}$ is the matrix of coefficients of the system. \hfill{\small\textbf{[2]}}
  \item Show that the system is consistent, and explain why the geometric arrangement is a sheaf of planes intersecting in a common straight line. \hfill{\small\textbf{[5]}}
\end{enumerate}

%── Q9 [GEO] ─────────────────────────────────────────────────────────────
\item \textbf{[GEO]} For the system of planes in Question~8:
\begin{enumerate}
  \item By setting $z = \lambda$, find expressions for $x$ and $y$ in terms of $\lambda$. \hfill{\small\textbf{[4]}}
  \item Hence write down a vector equation of the line of intersection in the form $\mathbf{r} = \mathbf{a} + \lambda \mathbf{d}$, where $\mathbf{a}$ and $\mathbf{d}$ are integer vectors. \hfill{\small\textbf{[2]}}
  \item Verify that every point on this line satisfies the equation of plane $\Pi_3$. \hfill{\small\textbf{[2]}}
\end{enumerate}

%── Q10 [GEO] ─────────────────────────────────────────────────────────────
\item \textbf{[GEO]} Three planes are defined by
\[
\begin{aligned}
\Pi_1 &: 2x - y + 3z = 4 \\
\Pi_2 &: 4x - 2y + 6z = 15 \\
\Pi_3 &: x + 2y - z = 1
\end{aligned}
\]
\begin{enumerate}
  \item Show that $\Pi_1$ and $\Pi_2$ are parallel and distinct. \hfill{\small\textbf{[2]}}
  \item Find the Cartesian and vector direction of the line of intersection of $\Pi_1$ and $\Pi_3$, and show that the intersection of $\Pi_2$ and $\Pi_3$ is a parallel line. \hfill{\small\textbf{[4]}}
  \item Describe fully the geometric arrangement of the three planes, stating the number of solutions to the simultaneous system. \hfill{\small\textbf{[2]}}
\end{enumerate}
