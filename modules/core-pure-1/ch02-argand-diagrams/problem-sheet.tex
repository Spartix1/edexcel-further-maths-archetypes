\CPTitleBlock{Argand Diagrams \& Loci}{Problem Sheet 2 \enspace$\cdot$\enspace Modulus-Argument Form, Geometric Loci \& Extremum Optimization}

\section*{Section A \quad Procedural \& Direct Foundations \hfill{\normalfont\footnotesize(Questions 1--3)}}
\begin{enumerate}

%── Q1 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} The complex numbers $z_1$ and $z_2$ are given by
\[
z_1 = -1 + i\sqrt{3} \quad \text{and} \quad z_2 = 2 - 2i
\]
\begin{enumerate}[label=(\alph*)]
  \item Find the exact modulus $|z_1|$ and the principal argument $\arg(z_1)$, giving the argument in radians as an exact multiple of $\pi$. \hfill{\textbf{[2 marks]}}
  \item Find the exact modulus $|z_2|$ and the principal argument $\arg(z_2)$, giving the argument in radians as an exact multiple of $\pi$. \hfill{\textbf{[2 marks]}}
  \item Write down the exact value of $\left|\dfrac{z_1}{z_2}\right|$ and evaluate $\arg\left(\dfrac{z_1}{z_2}\right)$ as an exact multiple of $\pi$. \hfill{\textbf{[2 marks]}}
\end{enumerate}

%── Q2 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} Two complex numbers $w_1$ and $w_2$ are defined in modulus-argument form by
\[
w_1 = 4\left(\cos\frac{5\pi}{6} + i\sin\frac{5\pi}{6}\right) \quad \text{and} \quad w_2 = 2\left(\cos\left(-\frac{\pi}{3}\right) + i\sin\left(-\frac{\pi}{3}\right)\right)
\]
\begin{enumerate}[label=(\alph*)]
  \item Express $w_1 w_2$ in modulus-argument form $r(\cos\theta + i\sin\theta)$, where $r > 0$ and $-\pi < \theta \le \pi$. \hfill{\textbf{[2 marks]}}
  \item Hence express $w_1 w_2$ in exact Cartesian form $a + bi$. \hfill{\textbf{[1 mark]}}
  \item Evaluate $(w_2)^6$, giving your answer as an exact integer. \hfill{\textbf{[2 marks]}}
\end{enumerate}

%── Q3 [GEO] ─────────────────────────────────────────────────────────────
\item \textbf{[GEO]} A point $P$ in the Argand plane represents the complex number $z$, which satisfies the equation
\[
|z - 3 + 4i| = 5
\]
\begin{enumerate}[label=(\alph*)]
  \item Describe geometrically the locus of $P$, stating the coordinates of any relevant points and the exact radius. \hfill{\textbf{[2 marks]}}
  \item Verify that the locus passes through the origin $O$. \hfill{\textbf{[1 mark]}}
  \item Find the Cartesian equation of this locus in the form $x^2 + y^2 + ax + by = 0$, where $a$ and $b$ are integers to be found. \hfill{\textbf{[2 marks]}}
\end{enumerate}

\end{enumerate}

\section*{Section B \quad Reverse Parameters \& Analysis \hfill{\normalfont\footnotesize(Questions 4--6)}}
\begin{enumerate}[resume]

%── Q4 [GEO] ─────────────────────────────────────────────────────────────
\item \textbf{[GEO]} The complex number $z$ satisfies the equation
\[
|z - 2 - 3i| = |z + 4 - i|
\]
\begin{enumerate}[label=(\alph*)]
  \item State the geometric definition of the locus of $z$ in terms of two fixed points $A$ and $B$. \hfill{\textbf{[2 marks]}}
  \item Find the Cartesian equation of this locus, giving your answer in the form $y = mx + c$, where $m$ and $c$ are integers. \hfill{\textbf{[3 marks]}}
\end{enumerate}

%── Q5 [GEO] ─────────────────────────────────────────────────────────────
\item \textbf{[GEO]} The complex number $z$ satisfies the half-line condition
\[
\arg(z + 1 - i\sqrt{3}) = \frac{\pi}{3}
\]
\begin{enumerate}[label=(\alph*)]
  \item Sketch this locus on an Argand diagram, clearly labelling the coordinates of the initial point and the direction of the ray. \hfill{\textbf{[2 marks]}}
  \item Find the Cartesian equation of the half-line in the form $y = mx + c$, stating clearly the domain of $x$ for which this equation is valid. Give $m$ and $c$ in exact surd form. \hfill{\textbf{[3 marks]}}
\end{enumerate}

%── Q6 [GEO] ─────────────────────────────────────────────────────────────
\item \textbf{[GEO]} On an Argand diagram, a closed region $R$ is defined by the simultaneous inequalities
\[
|z - 2i| \le 2, \quad \text{Re}(z) \ge 0, \quad \text{and} \quad \arg(z) \ge \frac{\pi}{4}
\]
\begin{enumerate}[label=(\alph*)]
  \item Sketch the boundary curves on a single Argand diagram and shade the region $R$. \hfill{\textbf{[3 marks]}}
  \item Show that the line $\arg(z) = \frac{\pi}{4}$ intersects the boundary $|z - 2i| = 2$ at the origin $O$ and at the point $P(2, 2)$. \hfill{\textbf{[2 marks]}}
  \item Hence determine the exact area of the shaded region $R$, giving your answer in terms of $\pi$. \hfill{\textbf{[3 marks]}}
\end{enumerate}

\end{enumerate}

\section*{Section C \quad Proof, Geometry \& Multi-step \hfill{\normalfont\footnotesize(Questions 7--10)}}
\begin{enumerate}[resume]

%── Q7 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} The complex number $z$ satisfies the locus equation
\[
|z - 5 - 12i| = 4
\]
\begin{enumerate}[label=(\alph*)]
  \item Find the exact minimum value and the exact maximum value of $|z|$. \hfill{\textbf{[3 marks]}}
  \item Find the unique complex number $z$ on the locus for which $|z|$ is a minimum, expressing your answer in the form $a + bi$, where $a$ and $b$ are rational numbers. \hfill{\textbf{[3 marks]}}
\end{enumerate}

%── Q8 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} A circle $C$ in the Argand plane is given by the equation
\[
|z - 4\sqrt{3} - 4i| = 4
\]
\begin{enumerate}[label=(\alph*)]
  \item Show that the distance from the origin $O$ to the centre of $C$ is 8, and deduce that the origin lies strictly outside the circle. \hfill{\textbf{[2 marks]}}
  \item Find the angle $\theta_0$ that the line joining the origin to the centre of $C$ makes with the positive real axis, giving your answer as an exact multiple of $\pi$. \hfill{\textbf{[1 mark]}}
  \item Two tangents are drawn from the origin to touch the circle $C$ at points $T_1$ and $T_2$. Prove that each tangent makes an angle of $\dfrac{\pi}{6}$ radians with the line joining the origin to the centre of $C$. \hfill{\textbf{[2 marks]}}
  \item Hence state the exact minimum and maximum possible values of $\arg(z)$ for points lying on the circle $C$, giving each value as an exact multiple of $\pi$. \hfill{\textbf{[2 marks]}}
\end{enumerate}

%── Q9 [SYN] ─────────────────────────────────────────────────────────────
\item \textbf{[SYN]} In the complex plane, two loci $L_1$ and $L_2$ are defined by
\[
L_1: \ |z - 4| = 4 \quad \text{and} \quad L_2: \ \arg(z) = \frac{\pi}{6}
\]
\begin{enumerate}[label=(\alph*)]
  \item Sketch both loci on the same Argand diagram. \hfill{\textbf{[2 marks]}}
  \item By converting each locus to its Cartesian form, show that the non-zero point of intersection of $L_1$ and $L_2$ satisfies the equation $\frac{4}{3}x^2 - 8x = 0$. \hfill{\textbf{[3 marks]}}
  \item Hence find the exact complex number $z$ representing the intersection point, giving your answer in the form $p + i\sqrt{q}$, where $p$ and $q$ are integers. \hfill{\textbf{[2 marks]}}
\end{enumerate}

%── Q10 [PRF] ────────────────────────────────────────────────────────────
\item \textbf{[PRF]} The complex variable $w$ is defined in terms of $z$ by the transformation
\[
w = \frac{z - 2}{z + 2}, \quad z \neq -2
\]
\begin{enumerate}[label=(\alph*)]
  \item Given that $|w| = 1$, prove algebraically that the locus of $z = x + iy$ in the complex plane is the imaginary axis ($x = 0$). \hfill{\textbf{[3 marks]}}
  \item Given instead that $\arg(w) = \dfrac{\pi}{2}$:
  \begin{enumerate}[label=(\roman*)]
    \item Explain the geometric meaning of $\arg\left(\dfrac{z - 2}{z + 2}\right) = \dfrac{\pi}{2}$ in terms of the angle subtended by the line segment joining $-2$ and $2$. \hfill{\textbf{[2 marks]}}
    \item Deduce that the locus of $z$ is a semicircle, stating its centre, exact radius, and any excluded boundary points. \hfill{\textbf{[3 marks]}}
  \end{enumerate}
\end{enumerate}

\end{enumerate}
