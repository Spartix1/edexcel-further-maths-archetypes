\CPTitleBlock{Core Pure 1: Complex Numbers}{Problem Sheet \#1 --- Cartesian Arithmetic, Conjugates, Roots of Polynomials \& Square Roots}
\noindent\textit{Instructions: Answer all questions. Fully justify your mathematical reasoning. Show all algebraic steps clearly.}

\section*{Section A \quad Procedural \& Direct Foundations \hfill{\normalfont\small(Questions 1--3)}}
\begin{enumerate}

%── Q1 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} The complex numbers $z_1$ and $z_2$ are given by
\[
z_1 = 3 - 2i \quad \text{and} \quad z_2 = 1 + 4i
\]
\begin{enumerate}[label=(\alph*)]
  \item Express $\dfrac{z_1}{z_2}$ in the form $x + iy$, where $x$ and $y$ are rational numbers. Show all your working clearly. \hfill{\textbf{[3 marks]}}
  \item Find the exact value of $|z_1|^2 + |z_2|^2$. \hfill{\textbf{[2 marks]}}
  \item Write down the complex conjugate $z_1^*$. \hfill{\textbf{[1 mark]}}
\end{enumerate}

%── Q2 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} The complex number $w$ is defined by $w = -2\sqrt{3} + 2i$.
\begin{enumerate}[label=(\alph*)]
  \item Find the exact value of the modulus $|w|$. \hfill{\textbf{[1 mark]}}
  \item Find the exact value of the principal argument $\arg(w)$, giving your answer in radians as an exact multiple of $\pi$. \hfill{\textbf{[2 marks]}}
  \item Hence, or otherwise, express $w^3$ in the form $a + bi$, where $a, b \in \mathbb{R}$. \hfill{\textbf{[2 marks]}}
\end{enumerate}

%── Q3 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} 
\begin{enumerate}[label=(\alph*)]
  \item Solve the quadratic equation
  \[
  2z^2 - 12z + 28 = 0
  \]
  giving your answers in the form $p \pm i\sqrt{q}$, where $p$ and $q$ are integers. \hfill{\textbf{[3 marks]}}
  \item Show both roots on a single, clearly labelled Argand diagram. \hfill{\textbf{[2 marks]}}
\end{enumerate}

\end{enumerate}

\section*{Section B \quad Reverse Parameters \& Analysis \hfill{\normalfont\small(Questions 4--6)}}
\begin{enumerate}[resume]

%── Q4 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} Given that $w = -7 + 24i$:
\begin{enumerate}[label=(\alph*)]
  \item Find the two square roots of $w$, giving your answers in the form $\pm(a + bi)$, where $a$ and $b$ are integers. You must show full algebraic working. \hfill{\textbf{[5 marks]}}
  \item Hence solve the equation $(z - 3i)^2 = -7 + 24i$. \hfill{\textbf{[2 marks]}}
\end{enumerate}

%── Q5 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} The complex number $z = x + iy$, where $x$ and $y$ are real numbers.
Given that
\[
2z + 3iz^* = 13 + 12i
\]
where $z^*$ denotes the complex conjugate of $z$, find the value of $x$ and the value of $y$. \hfill{\textbf{[6 marks]}}

%── Q6 [PRF/REV] ─────────────────────────────────────────────────────────
\item \textbf{[PRF]} The cubic polynomial $f(z)$ is defined by
\[
f(z) = 2z^3 - 7z^2 + 16z - 15
\]
Given that $z = 1 + 2i$ is a root of the equation $f(z) = 0$:
\begin{enumerate}[label=(\alph*)]
  \item Write down the second complex root of $f(z) = 0$. \hfill{\textbf{[1 mark]}}
  \item Find the third root of the equation. \hfill{\textbf{[3 marks]}}
  \item Explain why the non-real roots of any polynomial with real coefficients must occur in conjugate pairs. \hfill{\textbf{[2 marks]}}
\end{enumerate}

\end{enumerate}

\section*{Section C \quad Proof, Geometry \& Multi-step \hfill{\normalfont\small(Questions 7--10)}}
\begin{enumerate}[resume]

%── Q7 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} The quartic equation
\[
z^4 - 4z^3 + 16z^2 + pz + q = 0
\]
where $p$ and $q$ are real constants, has a root $z_1 = 1 - 3i$.
\begin{enumerate}[label=(\alph*)]
  \item Write down the conjugate root $z_2$. \hfill{\textbf{[1 mark]}}
  \item Find the values of the real constants $p$ and $q$. \hfill{\textbf{[4 marks]}}
  \item Hence determine the remaining roots of the equation. \hfill{\textbf{[2 marks]}}
\end{enumerate}

%── Q8 [GEO] ─────────────────────────────────────────────────────────────
\item \textbf{[GEO]} The points $P$, $Q$, and $O$ on an Argand diagram represent the complex numbers $z_1 = 4 + 2i$, $z_2 = -2 + 4i$, and $0$ (the origin) respectively.
\begin{enumerate}[label=(\alph*)]
  \item Find the exact values of $|z_1|$ and $|z_2|$. \hfill{\textbf{[2 marks]}}
  \item Show that $\angle POQ = \dfrac{\pi}{2}$. \hfill{\textbf{[2 marks]}}
  \item Triangle $OPQ$ is inscribed in a circle $C$. Find the complex number representing the centre of $C$ and state the exact radius of $C$. \hfill{\textbf{[3 marks]}}
\end{enumerate}

%── Q9 [SYN] ─────────────────────────────────────────────────────────────
\item \textbf{[SYN]} 
\begin{enumerate}[label=(\alph*)]
  \item Use the binomial theorem to expand $(2 - i)^4$, expressing your answer in the form $a + bi$, where $a, b \in \mathbb{Z}$. Show each term of your expansion. \hfill{\textbf{[4 marks]}}
  \item Given that $\arg(2 - i) = -\theta$, where $0 < \theta < \dfrac{\pi}{2}$, deduce the exact value of $\tan(4\theta)$. \hfill{\textbf{[3 marks]}}
\end{enumerate}

%── Q10 [SYN] ────────────────────────────────────────────────────────────
\item \textbf{[SYN]} 
\begin{enumerate}[label=(\alph*)]
  \item Solve the quadratic equation
  \[
  z^2 - (4 - i)z + (5 - 5i) = 0
  \]
  showing all stages of your algebraic working. Give your answers in the form $p + qi$, where $p, q \in \mathbb{Z}$. \hfill{\textbf{[6 marks]}}
  \item State, with a reason, whether the roots found in part (a) form a complex conjugate pair. \hfill{\textbf{[2 marks]}}
\end{enumerate}

\end{enumerate}
