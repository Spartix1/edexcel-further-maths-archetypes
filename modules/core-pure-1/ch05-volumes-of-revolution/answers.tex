\section*{Section A \quad Procedural \& Direct Foundations}
\begin{enumerate}

%── Q1 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} Standard rotation about the $x$-axis.
\ans{
$V = 30\pi$
}
\method{
M1: States volume formula $V = \pi \int_a^b y^2 \, dx$.
M1: Substitutes $y^2 = (2\sqrt{x})^2 = 4x$:
\[
V = \pi \int_1^4 4x \, dx = 4\pi \left[\frac{x^2}{2}\right]_1^4
\]
A1: Evaluates limits: $2\pi [4^2 - 1^2] = 2\pi(16 - 1) = 2\pi(15)$.
A1: $30\pi$.
}
\inv{
What is the mean value of the function $f(x) = y^2 = 4x$ over the interval $[1, 4]$? Notice $\bar{y^2} = \frac{1}{4-1}\int_1^4 4x dx = \frac{30}{3} = 10$. Thus the volume is simply $V = \pi \times \bar{y^2} \times \Delta x = \pi(10)(3) = 30\pi$. This illustrates Pappus's centroid theorem in action!
}

%── Q2 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} Standard rotation about the $y$-axis.
\ans{
(a) $x^2 = y^{2/3}$ \qquad (b) $V = \dfrac{93\pi}{5}$
}
\method{
\textbf{(a)} B1: $y = x^3 \implies x = y^{1/3} \implies x^2 = (y^{1/3})^2 = y^{2/3}$.\\
\textbf{(b)} M1: States $V = \pi \int_c^d x^2 \, dy = \pi \int_1^8 y^{2/3} \, dy$.
M1: Integrates: $\pi \left[\frac{y^{5/3}}{5/3}\right]_1^8 = \frac{3\pi}{5} \left[y^{5/3}\right]_1^8$.
A1: Evaluates: $8^{5/3} = (2)^5 = 32$ and $1^{5/3} = 1$, giving $\frac{3\pi}{5}(32 - 1) = \frac{93\pi}{5}$.
}
\inv{
If this same region were rotated about the $x$-axis instead, would the resulting solid have greater or smaller volume? Calculate $V_x = \pi \int_1^2 x^6 dx = \pi [\frac{x^7}{7}]_1^2 = \frac{127\pi}{7} \approx 18.14\pi$. Compare this with $V_y = 18.6\pi$.
}

%── Q3 [DIR] ─────────────────────────────────────────────────────────────
\item \textbf{[DIR]} Parametric rotation about the $x$-axis and trigonometric reduction.
\ans{
(a) Proof of integral $12\pi \int_0^{\pi/2} \sin^3\theta \, d\theta$ \qquad (b) $V = 8\pi$
}
\method{
\textbf{(a)} M1: $x = 3\cos\theta \implies \frac{dx}{d\theta} = -3\sin\theta$.
M1: Limits: when $x = 0$, $\theta = \frac{\pi}{2}$; when $x = 3$, $\theta = 0$.
M1: $V = \pi \int y^2 \frac{dx}{d\theta} d\theta = \pi \int_{\pi/2}^0 (2\sin\theta)^2 (-3\sin\theta) d\theta = 12\pi \int_0^{\pi/2} \sin^3\theta \, d\theta$.
A1: Fully justified integral.\\
\textbf{(b)} M1: Writes $\sin^3\theta = \sin\theta(1 - \cos^2\theta)$.
M1: Integrates: $\int_0^{\pi/2} (\sin\theta - \sin\theta\cos^2\theta) d\theta = \left[-\cos\theta + \frac{\cos^3\theta}{3}\right]_0^{\pi/2}$.
A1: $(0 - 0) - \left(-1 + \frac{1}{3}\right) = \frac{2}{3}$.
A1: $V = 12\pi \times \frac{2}{3} = 8\pi$.
}
\inv{
The curve is the upper-right quadrant of the ellipse $\frac{x^2}{3^2} + \frac{y^2}{2^2} = 1$. The full ellipsoid generated by rotating an ellipse of semi-axes $a$ and $b$ about the $x$-axis has volume $V = \frac{4}{3}\pi a b^2 = \frac{4}{3}\pi (3)(2^2) = 16\pi$. The quarter-ellipse generates exactly half of this ellipsoid: $\frac{1}{2}(16\pi) = 8\pi$!
}

\end{enumerate}

\section*{Section B \quad Reverse Parameters \& Analysis}
\begin{enumerate}[resume]

%── Q4 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} Parametric rotation about the $y$-axis with power terms.
\ans{
$V = \dfrac{1536\pi}{7}$
}
\method{
M1: $y = t^3 \implies \frac{dy}{dt} = 3t^2$.
M1: Limits: $y = 0 \implies t = 0$; $y = 8 \implies t^3 = 8 \implies t = 2$.
M1: $V = \pi \int x^2 \frac{dy}{dt} dt = \pi \int_0^2 (2t^2)^2 (3t^2) dt = \pi \int_0^2 (4t^4)(3t^2) dt = 12\pi \int_0^2 t^6 dt$.
A1: Integrates: $12\pi \left[\frac{t^7}{7}\right]_0^2 = \frac{12\pi}{7}(2^7 - 0) = \frac{12\pi(128)}{7}$.
A1: $\frac{1536\pi}{7}$.
}
\inv{
Eliminate the parameter $t$: since $x = 2t^2$ and $y = t^3$, $t = y^{1/3} \implies x = 2y^{2/3}$. Then $x^2 = 4y^{4/3}$. Integrate directly: $\pi \int_0^8 4y^{4/3} dy = 4\pi [\frac{3}{7}y^{7/3}]_0^8 = \frac{12\pi}{7}(8^{7/3}) = \frac{12\pi}{7}(128) = \frac{1536\pi}{7}$. Both methods produce the exact same fraction!
}

%── Q5 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} Hollow solid bounded between intersecting curves.
\ans{
$V = \dfrac{2\pi}{15}$
}
\method{
M1: Finds intersections: $x = x^2 \implies x(x - 1) = 0 \implies x = 0$ and $x = 1$.
M1: For $0 \le x \le 1$, $x \ge x^2$, so outer curve is $y = x$ and inner curve is $y = x^2$.
M1: Formula for hollow volume: $V = \pi \int_0^1 (y_{\text{outer}}^2 - y_{\text{inner}}^2) \, dx = \pi \int_0^1 (x^2 - x^4) \, dx$.
A1: Integrates: $\pi \left[\frac{x^3}{3} - \frac{x^5}{5}\right]_0^1 = \pi \left(\frac{1}{3} - \frac{1}{5}\right)$.
A1: $\frac{2\pi}{15}$.
}
\inv{
What common mistake do students make on hollow solids? Many candidates evaluate $\pi \int (y_{\text{outer}} - y_{\text{inner}})^2 dx = \pi \int (x - x^2)^2 dx = \frac{\pi}{30}$, which is strictly incorrect! Explain geometrically why $\pi(R^2 - r^2) \neq \pi(R - r)^2$.
}

%── Q6 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} Paraboloid of revolution about the $y$-axis.
\ans{
(a) $V = \pi \displaystyle\int_0^4 (4 - y) \, dy$ \qquad (b) $V = 8\pi$
}
\method{
\textbf{(a)} M1: Rearranges $y = 4 - x^2 \implies x^2 = 4 - y$.
A1: Limits on $y$: when $x = 0$, $y = 4$; when $y = 0$, $x = 2$. Integral is $\pi \int_0^4 (4 - y) \, dy$.\\
\textbf{(b)} M1: Integrates: $\pi \left[4y - \frac{y^2}{2}\right]_0^4$.
A1: Evaluates: $\pi \left(4(4) - \frac{16}{2}\right) = \pi(16 - 8) = 8\pi$.
}
\inv{
Compare this with the volume of the circumscribing cylinder: Radius $R = 2$, Height $H = 4$. Cylinder volume $V_{\text{cyl}} = \pi R^2 H = \pi(4)(4) = 16\pi$. Notice that the paraboloid occupies exactly half of the enclosing cylinder: $\frac{8\pi}{16\pi} = \frac{1}{2}$! (Archimedes proved this in 250 BC).
}

\end{enumerate}

\section*{Section C \quad Proof, Geometry \& Multi-step}
\begin{enumerate}[resume]

%── Q7 [CTX] ─────────────────────────────────────────────────────────────
\item \textbf{[CTX]} Container modelling and related rates of liquid filling.
\ans{
(a) Proof of total capacity $32\pi \text{ cm}^3$ \qquad (b) $\dfrac{dh}{dt} = \dfrac{1}{2} \text{ cm/s}$
}
\method{
\textbf{(a)} M1: $x = 2\sqrt{y} \implies x^2 = 4y$.
M1: Capacity $V = \pi \int_0^4 4y \, dy = \pi [2y^2]_0^4$.
A1: $\pi [2(16) - 0] = 32\pi \text{ cm}^3$.\\
\textbf{(b)} M1: Volume at depth $h$: $V(h) = \pi \int_0^h 4y \, dy = 2\pi h^2$.
M1: Differentiates with respect to $t$ using the chain rule:
\[
\frac{dV}{dt} = \frac{dV}{dh} \cdot \frac{dh}{dt} = 4\pi h \cdot \frac{dh}{dt}
\]
M1: Substitutes $\frac{dV}{dt} = 6\pi$ and $h = 3$:
\[
6\pi = 4\pi(3) \cdot \frac{dh}{dt} = 12\pi \cdot \frac{dh}{dt}
\]
A1: $\frac{dh}{dt} = \frac{6\pi}{12\pi} = \frac{1}{2} \text{ cm/s}$ (or $0.5 \text{ cm/s}$).
}
\inv{
Notice that $\frac{dh}{dt} = \frac{6\pi}{4\pi h} = \frac{3}{2h}$ is inversely proportional to depth. As the bowl fills, why does the rate of water level rise decelerate? (Because cross-sectional area increases linearly with height: $A(h) = \pi x^2 = 4\pi h$).
}

%── Q8 [REV] ─────────────────────────────────────────────────────────────
\item \textbf{[REV]} Reverse parameter determination for reciprocal curve.
\ans{
$k = 4$
}
\method{
M1: $V = \pi \int_1^4 y^2 \, dx = \pi \int_1^4 \left(\frac{k}{x}\right)^2 \, dx = \pi k^2 \int_1^4 x^{-2} \, dx$.
M1: Integrates: $\pi k^2 \left[-\frac{1}{x}\right]_1^4$.
A1: $\pi k^2 \left(-\frac{1}{4} - (-1)\right) = \frac{3\pi k^2}{4}$.
M1: Sets volume equal to $12\pi$:
\[
\frac{3\pi k^2}{4} = 12\pi \implies 3k^2 = 48 \implies k^2 = 16
\]
A1: Since $k > 0$, $k = 4$.
}
\inv{
What happens to the volume as the upper limit tends to infinity? Evaluate $V_\infty = \pi k^2 \int_1^\infty x^{-2} dx = \pi k^2 [-1/x]_1^\infty = \pi k^2$. Contrast this with the infinite surface area of Gabriel's Horn ($\int_1^\infty \frac{2\pi}{x} dx = \infty$)!
}

%── Q9 [SYN] ─────────────────────────────────────────────────────────────
\item \textbf{[SYN]} Trigonometric rotation and cylinder comparison.
\ans{
(a) $V = \pi$ \qquad (b) Geometric verification that $\pi < \dfrac{\pi^2}{2}$
}
\method{
\textbf{(a)} M1: $y^2 = (\sqrt{\cos x})^2 = \cos x$.
M1: $V = \pi \int_0^{\pi/2} \cos x \, dx = \pi [\sin x]_0^{\pi/2}$.
A1: $\pi [\sin(\pi/2) - \sin(0)] = \pi(1 - 0) = \pi$.\\
\textbf{(b)} B1: Volume of enclosing cylinder: $V_{\text{cyl}} = \pi r^2 h = \pi (1)^2 \left(\frac{\pi}{2}\right) = \frac{\pi^2}{2}$.
B1: Since $\pi \approx 3.1416$, $\frac{\pi^2}{2} \approx \frac{9.8696}{2} \approx 4.935 > 3.1416 = \pi$, confirming the solid is strictly contained within the cylinder.
}
\inv{
Notice how the square root cleanly eliminates the trigonometric exponent: $(\sqrt{\cos x})^2 = \cos x$. This is a favourite Edexcel technique to turn a difficult integration into an instantaneous one-step integral!
}

%── Q10 [SYN] ────────────────────────────────────────────────────────────
\item \textbf{[SYN]} Frustum of a cone comparison via calculus and geometry.
\ans{
(a) Calculus proof: $V = \dfrac{112\pi}{3}$ \qquad (b) Geometric formula verification
}
\method{
\textbf{(a)} M1: $y^2 = \left(2 + \frac{x}{2}\right)^2$.
M1: Integrates using reverse chain rule:
\[
\int_0^4 \left(2 + \frac{x}{2}\right)^2 \, dx = \left[\frac{(2 + x/2)^3}{3 \times (1/2)}\right]_0^4 = \frac{2}{3} \left[\left(2 + \frac{x}{2}\right)^3\right]_0^4
\]
M1: Evaluates limits: when $x = 4$, $2 + 2 = 4$, so $4^3 = 64$; when $x = 0$, $2^3 = 8$.
A1: $V = \pi \times \frac{2}{3}(64 - 8) = \frac{2\pi(56)}{3} = \frac{112\pi}{3}$.\\
\textbf{(b)} M1: At $x = 0$, $r_1 = 2$; at $x = 4$, $r_2 = 2 + 2 = 4$; height $h = 4 - 0 = 4$.
M1: Substitutes into formula:
\[
V_{\text{geom}} = \frac{1}{3}\pi(4)(2^2 + (2)(4) + 4^2) = \frac{4\pi}{3}(4 + 8 + 16)
\]
A1: $\frac{4\pi}{3}(28) = \frac{112\pi}{3}$, which perfectly matches the calculus result.
}
\inv{
Why does rotating a straight line $y = mx + c$ always generate a cone or frustum? Prove algebraically that $\pi \int_0^h (mx + c)^2 dx \equiv \frac{1}{3}\pi h(c^2 + c(mh+c) + (mh+c)^2)$, proving that calculus and Euclidean geometry are in absolute harmony!
}

\end{enumerate}
