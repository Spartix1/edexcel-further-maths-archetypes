\CPTitleBlock{Proof by Mathematical Induction}{Problem Sheet \#8 \enspace$\cdot$\enspace Summations, Divisibility, Matrix Powers \& Recurrence Relations}

\section*{Section A \quad Procedural \& Direct Foundations \hfill{\normalfont\footnotesize(Questions 1--3)}}
\begin{enumerate}
%── Q1 [PRF] ─────────────────────────────────────────────────────────────
\item \textbf{[PRF]} Prove by mathematical induction that, for all positive integers $n$,
\[
\sum_{r=1}^{n} r(r+1) = \frac{n(n+1)(n+2)}{3}\,.
\]
\hfill{\small\textbf{[5]}}

%── Q2 [PRF] ─────────────────────────────────────────────────────────────
\item \textbf{[PRF]} Prove by mathematical induction that, for all positive integers $n$,
\[
\sum_{r=1}^{n} r \cdot 2^r = 2 + (n-1) \cdot 2^{n+1}\,.
\]
\hfill{\small\textbf{[6]}}

%── Q3 [PRF] ─────────────────────────────────────────────────────────────
\item \textbf{[PRF]} Prove by mathematical induction that, for all positive integers $n$,
\[
\sum_{r=1}^{n} r \cdot r! = (n+1)! - 1\,.
\]
\hfill{\small\textbf{[5]}}

\end{enumerate}

\section*{Section B \quad Reverse Parameters \& Analysis \hfill{\normalfont\small(Questions 4--6)}}
\begin{enumerate}[resume]
%── Q4 [PRF] ─────────────────────────────────────────────────────────────
\item \textbf{[PRF]} Prove by mathematical induction that, for all positive integers $n$,
\[
5^n + 8n - 1 \quad \text{is divisible by 16.}
\]
\hfill{\small\textbf{[6]}}

%── Q5 [PRF] ─────────────────────────────────────────────────────────────
\item \textbf{[PRF]} Prove by mathematical induction that $3^{2n} - 1$ is divisible by $8$ for all positive integers $n$.
\hfill{\small\textbf{[5]}}

%── Q6 [PRF] ─────────────────────────────────────────────────────────────
\item \textbf{[PRF]} Prove by mathematical induction that $7^n + 4^n + 1$ is divisible by $6$ for all positive integers $n$.
\hfill{\small\textbf{[6]}}

\end{enumerate}

\section*{Section C \quad Proof, Geometry \& Multi-step \hfill{\normalfont\small(Questions 7--10)}}
\begin{enumerate}[resume]
%── Q7 [PRF] ─────────────────────────────────────────────────────────────
\item \textbf{[PRF]} Let $\mathbf{M} = \begin{pmatrix} 3 & -2 \\ 2 & -1 \end{pmatrix}$.
Prove by mathematical induction that, for all positive integers $n$,
\[
\mathbf{M}^n = \begin{pmatrix} 2n+1 & -2n \\ 2n & 1-2n \end{pmatrix}\,.
\]
\hfill{\small\textbf{[8]}}

%── Q8 [PRF] ─────────────────────────────────────────────────────────────
\item \textbf{[PRF]} A sequence is defined by $u_1 = 1$ and $u_{n+1} = 3u_n + 4$ for $n \geq 1$.
Prove by mathematical induction that $u_n = 3^n - 2$ for all positive integers $n$.
\hfill{\small\textbf{[6]}}

%── Q9 [PRF] ─────────────────────────────────────────────────────────────
\item \textbf{[PRF]} A sequence is defined by $u_1 = 1$, $u_2 = 5$, and $u_{n+2} = 5u_{n+1} - 6u_n$ for $n \geq 1$.
Prove by mathematical induction that $u_n = 3^n - 2^n$ for all positive integers $n$.
\hfill{\small\textbf{[8]}}

%── Q10 [PRF] ─────────────────────────────────────────────────────────────
\item \textbf{[PRF]} Prove by mathematical induction that $2^n > n^2$ for all integers $n \geq 5$.
\hfill{\small\textbf{[7]}}

\end{enumerate}
